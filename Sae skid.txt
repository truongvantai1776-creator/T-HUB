EggState.CarryChanged:Connect(function(arg)
	local carrying=type(arg)=="table" and arg.IsCarrying==true
	if carrying and arg.GuardDisabled==true then carrying=false end
	YState.GuessedDrop=false
	if carrying then YState.HeldSeen=os.clock() end
	if carrying and type(arg.Uid)=="string" then
		YState.Uid=arg.Uid
		local m=tonumber(arg.SpeedMultiplier)
		if m and m>0 then YState.Mult=m end
	end
	YState.Carrying=carrying
end)

	networking:FindFirstChild("RE/EggWorld/FieldEggRedeemVerdict").OnClientEvent:Connect(function()
		YState.Delivered=os.clock()
	end)

end)

local function Yroot()
	local c=localPlayer.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function YwalkSpeed()
	local c=localPlayer.Character
	local h=c and c:FindFirstChildOfClass("Humanoid")
	local s=h and h.WalkSpeed or 16
	local ok,r=pcall(function()
		local st=localPlayer:FindFirstChild("leaderstats")
		st=st and st:FindFirstChild("Speed")
		local util=require(ReplicatedStorage.Shared.Util.TreadmillUtil)
		return st and util.SpeedPowerToWalkSpeed(st.Value) or nil
	end)
	if ok and type(r)=="number" and r>0 then s=math.min(s,r) end
	return s
end

local function YpromptNear(pos,rad)
	local best,bd=nil,rad
	for _,ch in ipairs(workspace:GetChildren()) do
		if ch.Name=="SmartPromptPart" and ch:IsA("BasePart") then
			local p=ch:FindFirstChild("CarryAreaEgg")
			if p and p:IsA("ProximityPrompt") then
				local d=(ch.Position-pos).Magnitude
				if d<bd then best,bd=p,d end
			end
		end
	end
	return best
end

local function YheldByMe(uid)
	local c=localPlayer.Character
	if type(uid)~="string" or not c then return false end
	local egg=workspace:FindFirstChild(uid)
	if not egg then return false end
	for _,d in ipairs(egg:GetDescendants()) do
		if d:IsA("WeldConstraint") or d:IsA("JointInstance") then
			local ok,a,b=pcall(function() return d.Part0,d.Part1 end)
			if ok and ((a and a:IsDescendantOf(c)) or (b and b:IsDescendantOf(c))) then return true end
		end
	end
	return false
end

task.spawn(function()
	while true do
		task.wait(0.2)
		if not YState.Carrying then
			if YState.GuessedDrop and YheldByMe(YState.Uid) then

			end

		elseif os.clock()-YState.HeldSeen>0.8 then

		end
	end
end)

local function YeggPosition(uid)
	local ok,res=pcall(function()
		return networking:FindFirstChild("RF/EggWorld/AskFieldEggSnapshot"):InvokeServer()
	end)
	local recs=ok and res and res.Records
	if not recs then return nil end
	for _,rec in pairs(recs) do
		if rec.Uid==uid and typeof(rec.BottomCFrame)=="CFrame" then
			return rec.BottomCFrame.Position
		end
	end
	return nil
end

local function YfrozenCamera()
	local cam=workspace.CurrentCamera
	if not cam then return function() end end
	local ot=cam.CameraType
	local f=cam.CFrame
	pcall(function() cam.CameraType=Enum.CameraType.Scriptable cam.CFrame=f end)
	return function() pcall(function() cam.CameraType=ot end) end
end

local fpsGen=0
local function YfpsOn()
	if typeof(setfpscap)~="function" then return end
	fpsGen+=1
	local mine=fpsGen
	task.spawn(function()
		local hi=true
		local st=os.clock()
		while fpsGen==mine and os.clock()-st<60 do
			pcall(setfpscap,hi and FPS[1] or FPS[2])
			hi=not hi
			task.wait(FPS[3])
		end
	end)
end
local function YfpsOff()
	fpsGen+=1
	if typeof(setfpscap)=="function" then pcall(setfpscap,240) end
end

local Yclone=nil
local function YdropClone()
	local c=Yclone
	Yclone=nil
	if c then pcall(function() c:Destroy() end) end
end
local function YpostClone()
	YdropClone()
	local c=localPlayer.Character
	if not c then return end
	local w=c.Archivable
	c.Archivable=true
	local copy=c:Clone()
	c.Archivable=w
	if copy then
		for _,d in ipairs(copy:GetDescendants()) do
			if d:IsA("LuaSourceContainer") or d:IsA("Humanoid") then
				pcall(function() d:Destroy() end)
			elseif d:IsA("BasePart") then
				d.Anchored=true
				d.CanCollide=false
				d.CanTouch=false
				d.CanQuery=false
			end
		end
		copy.Name="Clone"
		copy.Parent=workspace
		Yclone=copy
	end
end

local function Yplace(pos)
	local r=Yroot()
	if not r then return end
	pcall(function()
		r.CFrame=CFrame.new(pos)*CFrame.Angles(0,math.rad(90),0)
		r.AssemblyLinearVelocity=Vector3.zero
		r.AssemblyAngularVelocity=Vector3.zero
	end)
end

local dangerCache,dangerAt={},0
local function Ydangers()
	if os.clock()-dangerAt<1 then return dangerCache end
	dangerAt=os.clock()
	local list={}
	local function add(inst)
		local ok,cf,size=pcall(function()
			if inst:IsA("Model") then return inst:GetBoundingBox()
			elseif inst:IsA("BasePart") then return inst.CFrame,inst.Size end
		end)
		if ok and cf and size then
			local half=Vector3.new(math.abs(size.X),0,math.abs(size.Z))*0.5
			local rot=(cf-cf.Position):VectorToWorldSpace(half)
			local rx=math.max(math.abs(rot.X),half.X,half.Z)
			local rz=math.max(math.abs(rot.Z),half.X,half.Z)
			list[#list+1]={MinX=cf.Position.X-rx,MaxX=cf.Position.X+rx,MinZ=cf.Position.Z-rz,MaxZ=cf.Position.Z+rz}
		end
	end
	local function bad(name)
		if name=="ScrambleLocalVisuals" or name=="DrScrambleEvent" then return false end
		name=string.lower(name)
		return string.find(name,"portal",1,true) or string.find(name,"teleport",1,true)
			or string.find(name,"mech",1,true) or string.find(name,"arena",1,true)
			or string.find(name,"scramble",1,true)
	end
	for _,ch in ipairs(workspace:GetChildren()) do
		if (ch:IsA("Model") or ch:IsA("BasePart") or ch:IsA("Folder")) and bad(ch.Name) then
			if ch:IsA("Folder") then
				for _,inner in ipairs(ch:GetChildren()) do add(inner) end
			else add(ch) end
		end
	end
	local b=workspace:FindFirstChild("World")
	b=b and b:FindFirstChild("Build")
	if b then
		for _,ch in ipairs(b:GetChildren()) do
			if bad(ch.Name) then
				for _,inner in ipairs(ch:GetChildren()) do add(inner) end
			end
		end
	end
	dangerCache=list
	return list
end

local function Yavoid(from,to)
	for _,d in ipairs(Ydangers()) do
		local x0,x1,z0,z1=d.MinX-12,d.MaxX+12,d.MinZ-12,d.MaxZ+12
		local inside=from.X>=x0 and from.X<=x1 and from.Z>=z0 and from.Z<=z1
		if not inside then
			local t0,t1,hit=0,1,true
			for _,axis in ipairs({{from.X,to.X-from.X,x0,x1},{from.Z,to.Z-from.Z,z0,z1}}) do
				local pos,delta,lo,hi=axis[1],axis[2],axis[3],axis[4]
				if math.abs(delta)<1e-6 then
					if pos<lo or pos>hi then hit=false end
				else
					local ta,tb=(lo-pos)/delta,(hi-pos)/delta
					if ta>tb then ta,tb=tb,ta end
					t0,t1=math.max(t0,ta),math.min(t1,tb)
					if t0>t1 then hit=false end
				end
			end
			if hit then
				local zl,zh=z0-2,z1+2
				local z=math.abs(from.Z-zl)<=math.abs(from.Z-zh) and zl or zh
				if z<-440 or z>-290 then z=z==zl and zh or zl end
				local x=math.abs(from.X-x0)<=math.abs(from.X-x1) and x0 or x1
				if math.abs(from.Z-z)<3 then
					x=math.abs(to.X-x0)<=math.abs(to.X-x1) and x0 or x1
				end
				return Vector3.new(x,to.Y,z)
			end
		end
	end
	return to
end

local function YrunToEgg(uid,egg)
	local lineX=YlineInfo()
	local home=YhomePoint()
	local ch=localPlayer.Character
	local h=ch and ch:FindFirstChildOfClass("Humanoid")
	if h then
		h.PlatformStand=false
		if ch:FindFirstChildWhichIsA("Tool") then
			pcall(function() h:UnequipTools() end)
		end
	end
	local r=Yroot()
	if not r then return false end
	local stage="field"
	if r.Position.X<lineX-2 and Vector3.new(r.Position.X-home.X,0,r.Position.Z-home.Z).Magnitude>20 then
		stage="safe"
	end
	local started,lastCheck,lastPos,lastTake=os.clock(),os.clock(),r.Position,0
	while os.clock()-started<120 and not YState.Cancel do
		r=Yroot()
		if not r then return false end
		local flatEgg=Vector3.new(egg.X-r.Position.X,0,egg.Z-r.Position.Z)
		if stage=="field" and flatEgg.Magnitude<=2.5 then break end
		local target=egg
		if stage=="safe" then
			if Vector3.new(home.X-r.Position.X,0,home.Z-r.Position.Z).Magnitude<=6 then stage="field"
			else target=home end
		end
		local wp=Yavoid(r.Position,target)
		local flat=Vector3.new(wp.X-r.Position.X,0,wp.Z-r.Position.Z)
		local unit=flat.Magnitude>0.01 and flat.Unit or Vector3.zero
		local speed=math.max(YwalkSpeed()*CFG.SpeedCap,8)
		local v=unit*math.min(speed,flat.Magnitude/0.05)
		pcall(function()
			r.AssemblyLinearVelocity=Vector3.new(v.X,r.AssemblyLinearVelocity.Y,v.Z)
			if h and unit.Magnitude>0 then h:Move(unit,false) end
		end)
		if os.clock()-lastCheck>=1.5 then
			if (r.Position-lastPos).Magnitude<3 and h and flatEgg.Magnitude>15 then
				pcall(function() h.Jump=true end)
			end
			lastPos,lastCheck=r.Position,os.clock()
		end
		if stage=="field" and flatEgg.Magnitude<=9 and os.clock()-lastTake>0.1 then
			lastTake=os.clock()
			YtakeEgg(uid)
		end
		RunService.Heartbeat:Wait()
	end
	r=Yroot()
	if r then
		pcall(function()
			r.AssemblyLinearVelocity=Vector3.new(0,r.AssemblyLinearVelocity.Y,0)
			if h then h:Move(Vector3.zero,false) end
		end)
	end
	return YState.Carrying or (r~=nil and Vector3.new(egg.X-r.Position.X,0,egg.Z-r.Position.Z).Magnitude<=6)
end

local function YeggNow(uid,cache)
	local node=workspace:FindFirstChild(uid)
	local slots=workspace:FindFirstChild("AreaEggSlotsClient")
	node=node or (slots and slots:FindFirstChild(uid))
	if node then
		local ok,pos=pcall(function() return node:GetPivot().Position end)
		if ok and pos then cache.Pos,cache.At=pos,os.clock() return pos end
	end
	if os.clock()-cache.At>=0.5 then
		cache.At=os.clock()
		cache.Pos=YeggPosition(uid) or cache.Pos
	end
	return cache.Pos
end

local function Ygrab(uid,timeout)
	local cache={At=0}
	local waited,since=0,1
	while not YState.Carrying and waited<timeout and not YState.Cancel do
		local r=Yroot()
		local egg=YeggNow(uid,cache)
		local dt=math.max(RunService.Heartbeat:Wait(),1/240)
		waited+=dt
		since+=dt
		if r and egg then
			local delta=egg-r.Position
			if delta.Magnitude>2 then
				local pace=math.max(YwalkSpeed()*CFG.SpeedCap,16)
				local v=delta/math.max(0.08,dt)
				if v.Magnitude>pace then v=v.Unit*pace end
				pcall(function()
					r.AssemblyLinearVelocity=v+Vector3.new(0,workspace.Gravity*dt*0.5,0)
					r.AssemblyAngularVelocity=Vector3.zero
				end)
			end
			if since>=CFG.GrabInterval then
				since=0
				local prompt=YpromptNear(egg-Vector3.new(0,3,0),10)
				if prompt and typeof(fireproximityprompt)=="function" then
					pcall(function() prompt.HoldDuration=0 end)
					pcall(fireproximityprompt,prompt)
				end
				task.spawn(YtakeEgg,uid)
			end
		elseif since>=CFG.GrabInterval then
			since=0
			task.spawn(YtakeEgg,uid)
		end
	end
	local r=Yroot()
	if r then pcall(function() r.AssemblyLinearVelocity=Vector3.zero end) end
	return YState.Carrying and YState.Uid==uid
end

local function Yregrab(uid)
	if YState.Carrying then return true end
	local r=Yroot()
	local egg=YeggPosition(uid)
	if r and egg and Vector3.new(r.Position.X-egg.X,0,r.Position.Z-egg.Z).Magnitude>CFG.RegrabFar then
		Yplace(egg+Vector3.new(0,3,0))
	end
	return Ygrab(uid,3)
end

local function YgroundY(pos,fallback)
	local y=fallback
	pcall(function()
		local p=RaycastParams.new()
		p.FilterType=Enum.RaycastFilterType.Exclude
		p.FilterDescendantsInstances={localPlayer.Character}
		p.IgnoreWater=true
		local hit=workspace:Raycast(pos+Vector3.new(0,60,0),Vector3.new(0,-140,0),p)
		if hit and hit.Material~=Enum.Material.Water and math.abs(hit.Position.Y-fallback)<40 then
			y=hit.Position.Y+3.5
		end
	end)
	return y
end

local function YcarrySpeed(dist)
	local ws=YwalkSpeed()
	local base=ws*math.min(CFG.CarryRatio,CFG.SpeedCap)*YState.Mult
	local fast=base*1.5
	local excess=5.5*base
	local limit=fast
	if dist and dist>excess then
		limit=math.min(fast,base*dist/(dist-excess))
	end
	return math.min(math.max(math.min(base*CFG.EasyRatio,limit),base),math.max(ws*CFG.SpeedCap,base))
end

local function YrunHome(lineX,laneZ)
	local started=os.clock()
	local home=YhomePoint()
	local checkpoint=lineX-7
	local height=CFG.Height
	local share=math.clamp(CFG.ClimbShare,0.1,0.9)
	local descent=height*math.sqrt(1-share*share)/share
	local ch=localPlayer.Character
	local h=ch and ch:FindFirstChildOfClass("Humanoid")
	if h then h.PlatformStand=false end
	local r=Yroot()
	if r and ch and height>0.5 and home.Y+height-2>r.Position.Y then
		pcall(function()
			ch:PivotTo(CFrame.new(Vector3.new(r.Position.X,home.Y+height,r.Position.Z))*r.CFrame.Rotation)
			r.AssemblyLinearVelocity=Vector3.zero
			r.AssemblyAngularVelocity=Vector3.zero
		end)
	end
	local start=Yroot()
	local speed=YcarrySpeed(start and (Vector3.new(start.Position.X-home.X,0,start.Position.Z-home.Z).Magnitude+math.max(0,height)*2) or nil)
	local last=os.clock()
	local timeout=0
	while YState.Carrying and YState.Delivered<started and not YState.Cancel and timeout<25 do
		r=Yroot()
		if not r then return false end
		local now=os.clock()
		local dt=math.max(now-last,1/240)
		last=now
		timeout+=dt
		local toHome=r.Position.X<=checkpoint+2
		local target=toHome and home or Vector3.new(checkpoint,home.Y,laneZ)
		if toHome and Vector3.new(home.X-r.Position.X,0,home.Z-r.Position.Z).Magnitude<2 then break end
		local aim=Yavoid(r.Position,target)
		local flat=Vector3.new(aim.X-r.Position.X,0,aim.Z-r.Position.Z)
		local remaining=toHome and 0 or math.max(0,r.Position.X-checkpoint)
		local wantY=home.Y+height
		if toHome or remaining<=descent then
			wantY=home.Y+height*math.clamp(remaining/math.max(descent,1),0,1)
		end
		local vy=math.clamp((wantY-r.Position.Y)/0.12,-speed*share,speed*share)
		local horiz=math.sqrt(math.max(speed*speed-vy*vy,0))
		local v=flat.Magnitude>0.01 and flat.Unit*math.min(horiz,flat.Magnitude/0.05) or Vector3.zero
		pcall(function() r.AssemblyLinearVelocity=Vector3.new(v.X,vy,v.Z) end)
		RunService.Heartbeat:Wait()
	end
	if h then pcall(function() h:Move(Vector3.zero,false) end) end
	local settle=0
	while settle<2 and YState.Delivered<started and YState.Carrying and not YState.Cancel do
		settle+=RunService.Heartbeat:Wait()
	end
	if YState.Carrying and YState.Delivered<started and not YState.Cancel then
		task.wait(0.2)
		YdropEgg()
		local w=0
		while YState.Delivered<started and w<1 do
			w+=RunService.Heartbeat:Wait()
		end
	end
	return YState.Delivered>=started
end

local function YinstantTP(uid)
	local started=os.clock()
	local lineX,lineY=YlineInfo()
	local r=Yroot()
	if not r then return false end
	local laneZ=math.clamp(r.Position.Z,-425,-300)
	local landing=Vector3.new(lineX+CFG.LandOffset,lineY+3.35,laneZ)
	local hopStep=math.max(YwalkSpeed()*CFG.HopRatio,CFG.HopMin)
	local hopY=r.Position.Y+CFG.HopLift
	local x=r.Position.X
	local retries=0
	local releaseCamera=YfrozenCamera()
	YState.ReleaseCamera=releaseCamera
	pcall(YpostClone)
	YfpsOn()
	while x-hopStep>landing.X and not YState.Cancel do
		if not YState.Carrying then
			if not Yregrab(uid) then break end
			local cur=Yroot()
			if cur then x=math.min(x,cur.Position.X) end
			continue
		end
		local nextX=x-hopStep
		local pulledBefore=YState.PulledAt
		local held=0
		while held<CFG.HopGap do
			Yplace(Vector3.new(nextX,hopY,laneZ))
			held+=RunService.Heartbeat:Wait()
		end
		local check=Yroot()
		local pulled=YState.PulledAt>pulledBefore or (check~=nil and (check.Position.X-nextX>10 or check.AssemblyLinearVelocity.Magnitude>150))
		if pulled and YState.Carrying and not YState.Cancel then
			retries+=1
			CFG.HopGap=math.min(CFG.HopGapMax,CFG.HopGap+0.02)
			if retries>CFG.HopRetries then break end
			local settle=0
			while settle<0.12 and not YState.Cancel do
				Yplace(Vector3.new(x,hopY,laneZ))
				settle+=RunService.Heartbeat:Wait()
			end
		else
			x=nextX
			CFG.HopGap=math.max(CFG.HopGapMin,CFG.HopGap-0.01)
		end
	end
	landing=Vector3.new(landing.X,YgroundY(landing,landing.Y),landing.Z)
	local function land()
		local cur=Yroot()
		if cur and cur.Position.X<=landing.X+1 and cur.Position.Y>landing.Y-25 then
			pcall(function() cur.AssemblyLinearVelocity=Vector3.zero end)
			return
		end
		Yplace(landing)
	end
	land()
	for attempt=1,3 do
		local settle=0
		while settle<CFG.LandSettle and not YState.Cancel do
			settle+=RunService.Heartbeat:Wait()
		end
		local landed=Yroot()
		if YState.Carrying and landed and (landed.Position.X-landing.X>12 or landed.Position.Y<landing.Y-25) then
			land()
		else break end
	end
	YdropClone()
	if YState.Carrying and not YState.Cancel then
		local delay=0
		while delay<CFG.DropDelay do
			delay+=RunService.Heartbeat:Wait()
		end
		YdropEgg()
		local w=0
		while YState.Carrying and w<1 and not YState.Cancel do
			w+=RunService.Heartbeat:Wait()
		end
		local cur=Yroot()
		if cur then
			pcall(function()
				cur.AssemblyLinearVelocity=Vector3.zero
				cur.AssemblyAngularVelocity=Vector3.zero

tbl2.integrityRepairRecord=function(arg,arg2)
	local inst=arg and arg.instance

	end

		local v=arg.properties[arg2]
		if v~=nil then
			local ok,r=pcall(function() return inst[arg2] end)

	for k,prop in pairs(arg.properties) do

tbl2.formatNumber=function(arg)
	local n13=tonumber(arg) or 0
	local function fn2(a2)
		return (string.format("%.2f",a2):gsub("(%..-)0+$","%1"):gsub("%.$",""))
	end
	local n14=math.abs(n13)
	for _,v16 in ipairs({{1e12,"T"},{1e9,"B"},{1000000,"M"},{1000,"K"}}) do
		if n14>=v16[1] then
			local v17=v16[2]
			return fn2(n13/v16[1])..v17
		end
	end
	return tostring(math.floor(n13+0.5))
end

tbl2.getScaleMultiplier=function(arg)
	local n13=tonumber(arg) or 1
	if n13<=5 then return n13^1.85 end
	return (n13/5)^1.2*19.637875755794113
end

tbl2.getEggData=function(arg)
	local v16=Assets.Directory[arg.AssetCategory]
	if not v16 then return nil end
	local n13=tonumber(v16.EarningRate) or 0
	local n14=tonumber(arg.AssetScale) or 1
	local f2=arg.Mutations and #arg.Mutations>0
	local n15=1
	if f2 then
		if Mutations and type(Mutations.EarningsFor)=="function" then
			local ok,r=pcall(function() return Mutations.EarningsFor(arg.Mutations) end)
			if ok then n15=tonumber(r) or 1 end
		end
	end
	local n16=n13*tbl2.getScaleMultiplier(n14)*n15
	local c=Color3.fromRGB(170,190,178)
	local n17=0
	local s2=""
	if v16.Rarity then
		n17=v16.Rarity.RarityNumber or 0
		s2=v16.Rarity.DisplayName or v16.Rarity.Name or ""
		c=tbl2.getOfficialRarityColor(s2)
	end
	return {
		Uid=arg.Uid,Category=arg.AssetCategory,
		Name=v16.DisplayName or arg.AssetCategory,
		Generation=math.round(n16),
		Position=arg.BottomCFrame.Position,
		BottomCFrame=arg.BottomCFrame,
		Rarity=n17,RarityName=s2,RarityColor=c,
	}
end

tbl2.refreshEggs=function(arg)
	local now=tick()
	if not arg and now-tbl4.lastRefresh<tbl3.REFRESH_INTERVAL then return tbl4.eggs end
	local ok,r=pcall(function() return tbl.AskFieldEggSnapshot:InvokeServer() end)
	if not ok then return tbl4.eggs end
	local eggs=tbl4.eggs
	table.clear(eggs)
	if r and r.Records then
		for _,rec in ipairs(r.Records) do
			if rec.State=="Slot" or rec.State=="Dropped" then
				local v16=tbl2.getEggData(rec)
				if v16 then eggs[#eggs+1]=v16 end
			end
		end
	end
	table.sort(eggs,function(a2,a3) return a2.Generation>a3.Generation end)
	local fe=tbl4.filteredEggs
	table.clear(fe)
	for _,egg in ipairs(eggs) do
		if tbl2.eggMatchesRarityFilter(egg) then fe[#fe+1]=egg end
	end
	tbl4.bestEgg=tbl4.filteredEggs[1]
	tbl4.lastRefresh=now
	return tbl4.eggs
end

tbl2.getActiveTargetEgg=function()
	if tbl4.selectedEgg then
		for _,fe in ipairs(tbl4.filteredEggs) do
			if fe.Uid==tbl4.selectedEgg.Uid then
				tbl4.selectedEgg=fe
				tbl4.selectedUid=fe.Uid
				return fe
			end
		end
		tbl4.selectedEgg=nil
		tbl4.selectedUid=nil
	end
	return tbl4.bestEgg
end

		tbl2.updateToggle()
		return
	end

	if not tbl2.canLeaveForSteal() then
		tbl4.enabled=true
		tbl4.waitingForDay=true
		tbl2.updateToggle()
		tbl2.showToast(tbl2.isDaytime() and "Waiting for gate" or "Waiting for day")
		return
	end

	tbl4.enabled=true
	tbl4.waitingForDay=false
	tbl4.runLoop=tbl4.loopEnabled
	tbl2.updateToggle()

	task.spawn(function()
		while tbl4.enabled do
			if not tbl2.canLeaveForSteal() then
				tbl4.waitingForDay=true
				tbl2.updateToggle()
				while tbl4.enabled and not tbl2.canLeaveForSteal() do
					task.wait(1)
				end
				if not tbl4.enabled then break end
				tbl4.waitingForDay=false
				tbl2.updateToggle()
			end

			tbl2.refreshEggs(true)
			tbl2.updateBestDisplay()
			local target=tbl2.getActiveTargetEgg()
			if not target then
				task.wait(0.5)
				continue
			end

			tbl4.targetEgg=target
			tbl4.heldEggUid=target.Uid

			_G.yslemSteal(target.Uid,function()
				tbl4.targetEgg=nil
				tbl4.heldEggUid=nil
			end)

			while _G.yslemBusy() and tbl4.enabled do
				task.wait(0.2)
			end

			tbl4.targetEgg=nil
			tbl4.heldEggUid=nil

			if not tbl4.enabled then break end
			if not tbl4.loopEnabled then break end
			task.wait(0.2)
		end
		tbl4.enabled=false
		tbl4.waitingForDay=false
		tbl4.runLoop=false
		tbl2.updateToggle()
	end)
end)
